\begin{tikzpicture}[x=1cm,y=1cm,etq/.style={etiq,fill=white,inner sep=1.5pt}]
\def\sa{3.6}\def\sb{1.9}
\draw[gris,thick] (0,0) circle (1);
\draw[acero,very thick] (0,0) ellipse ({\sa} and {\sb});
\draw[eje] (-4.4,0) -- (4.7,0) node[right,etq]{$\sigma_1$};
\draw[eje] (0,-2.5) -- (0,2.9) node[above,etq]{$\sigma_2$};

\pgfmathsetmacro{\nx}{cos(\ang)}\pgfmathsetmacro{\ny}{sin(\ang)}
\pgfmathsetmacro{\tx}{\sa*\nx}\pgfmathsetmacro{\ty}{\sb*\ny}
\draw[vnormal] (0,0) -- (\nx,\ny);
\node[etq,text=verde] at (-0.28,0.95) {$\vec n$};
\draw[vtension] (0,0) -- (\tx,\ty) node[above right,etq]{$\vec t$};
\draw[dashed,tinta!60] (\tx,0) -- (\tx,\ty) -- (0,\ty);
\node[etq,text=acero,anchor=west] at (0.4,-2.85) {lugar de las puntas de $\vec t$};
